create table cliente (
	codcliente int PRIMARY KEY AUTO_INCREMENT,
	cliente varchar(255) not null,
	email varchar(100) UNIQUE not null, 
	cpf varchar(255) UNIQUE not null);

create table quarto (
	codquarto int PRIMARY KEY AUTO_INCREMENT,
	quarto varchar(255) not null,
	andar varchar(100) UNIQUE not null, 
	tipo varchar(100) UNIQUE not null);

create table servico (
	codservico int PRIMARY KEY AUTO_INCREMENT,
	servico varchar(255) not null,
	qtde int not null, 
	valor decimal(7,2) not null,
	tipo varchar(50) not null);
    
create table hospedagem (
	idhospedagem int PRIMARY KEY AUTO_INCREMENT,
	dataentrada date not null,
	datasaida date, 
	horaentrada time not null,
    horasaida time,
	totalhospedagem decimal(7,2),
	codcli int,
	codquarto int,
    FOREIGN KEY (codcli) REFERENCES cliente(codcliente),
    FOREIGN KEY (codquarto) REFERENCES quarto(codquarto));
