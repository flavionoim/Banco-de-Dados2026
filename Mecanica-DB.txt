CREATE TABLE mecanico 
( 
 nome_mec VARCHAR(255) NOT NULL,  
 id_mec INT PRIMARY KEY AUTO_INCREMENT NOT NULL,  
 telefone_mec VARCHAR(20) NOT NULL,  
 data_admissao_mec DATE NOT NULL,  
 cpf VARCHAR(20) UNIQUE NOT NULL,
 email VARCHAR(100) UNIQUE NOT NULL
); 

CREATE TABLE servicos 
( 
 cod_servico INT PRIMARY KEY AUTO_INCREMENT,
 nome_servico VARCHAR(255) NOT NULL,  
 descricao_servico VARCHAR(255),  
 tempo_medio_servico VARCHAR(50)
); 

CREATE TABLE cliente 
( 
 id_cliente INT PRIMARY KEY AUTO_INCREMENT,
 nome_cliente VARCHAR(255) NOT NULL,
 tel_cliente VARCHAR(20),  
 cidade_cliente VARCHAR(100),  
 email_cliente VARCHAR(100)
); 

CREATE TABLE veiculo 
( 
 placa_veic VARCHAR(20) PRIMARY KEY,
 marca_veic VARCHAR(50),  
 modelo_veic VARCHAR(100),  
 ano_veic INT,  
 cor_veic VARCHAR(50),  
 idcliente INT,
 CONSTRAINT FK_Veiculo_Cliente FOREIGN KEY(idcliente) REFERENCES cliente (id_cliente)
); 

CREATE TABLE ordemservico 
( 
 id_ordem INT PRIMARY KEY AUTO_INCREMENT,
 data_entrada DATE,  
 data_saida DATE,  
 problema_relatado VARCHAR(255),  
 valor_total DECIMAL(10,2),  
 status VARCHAR(50),  
 idveiculo VARCHAR(20),
 CONSTRAINT FK_OS_Veiculo FOREIGN KEY(idveiculo) REFERENCES veiculo (placa_veic)
); 

CREATE TABLE servicos_mecanicos 
( 
 codserv_mec INT PRIMARY KEY AUTO_INCREMENT,
 datainicio_servico DATE NOT NULL,  
 datafinalizacao_servico DATE NOT NULL,  
 valorunitario DECIMAL(10,2) NOT NULL,  
 observacao VARCHAR(255) NULL,  
 id_ordem INT,  
 id_mec INT,  
 cod_servico INT,  
 finalizado BIT NOT NULL,
 CONSTRAINT FK_SM_Ordem FOREIGN KEY(id_ordem) REFERENCES ordemservico (id_ordem),
 CONSTRAINT FK_SM_Mecanico FOREIGN KEY(id_mec) REFERENCES mecanico (id_mec),
 CONSTRAINT FK_SM_Servico FOREIGN KEY(cod_servico) REFERENCES servicos (cod_servico)
);
