CREATE TABLE tatuador 
( 
 idTatuador INT PRIMARY KEY,  
 idsenioridade INT,  
 nomeTatuador INT,  
 email INT,  
 telefone INT,  
 dataContratacao INT,  
 cpf INT,  
); 

CREATE TABLE estilo 
( 
 idEstilo INT PRIMARY KEY,  
 estilo INT,  
); 

CREATE TABLE agendamento 
( 
 idAgendamento INT PRIMARY KEY,  
 idtatuador INT,  
 idcliente INT,  
 status INT,  
 dataagendamento INT,  
 valorTatuagem INT,  
 horaAgendamento INT,  
 desenhoAprovado INT,  
 descricaoTatuagem INT,  
); 

CREATE TABLE cliente 
( 
 idCliente INT PRIMARY KEY,  
 uf INT,  
 dataCadastro INT,  
 cidade INT,  
 cpf INT,  
 nome INT,  
 telefone INT,  
); 

CREATE TABLE senioridade 
( 
 codsenioridade INT PRIMARY KEY,  
 nome INT,  
); 

CREATE TABLE possui 
( 
 idTatuador INT PRIMARY KEY,  
 idEstilo INT PRIMARY KEY,  
 dataEstilo INT,  
); 

ALTER TABLE tatuador ADD FOREIGN KEY(idsenioridade) REFERENCES senioridade (idsenioridade)
ALTER TABLE agendamento ADD FOREIGN KEY(idtatuador) REFERENCES tatuador (idtatuador)
ALTER TABLE agendamento ADD FOREIGN KEY(idcliente) REFERENCES cliente (idcliente)
ALTER TABLE possui ADD FOREIGN KEY(idTatuador) REFERENCES tatuador (idTatuador)
ALTER TABLE possui ADD FOREIGN KEY(idEstilo) REFERENCES estilo (idEstilo)
